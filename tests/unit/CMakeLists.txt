# m2022_unit_test(<name> <sources...> [LINK <libs...>]); links m2022_common unless LINK is given.
function(m2022_unit_test name)
    cmake_parse_arguments(ARG "" "" "LINK" ${ARGN})
    add_executable(${name} ${ARG_UNPARSED_ARGUMENTS})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(ARG_LINK)
        target_link_libraries(${name} PRIVATE ${ARG_LINK})
    else()
        target_link_libraries(${name} PRIVATE m2022_common)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "unit")
endfunction()

m2022_unit_test(test_version test_version.c)
m2022_unit_test(test_json test_json.c)
m2022_unit_test(test_ieee1284 test_ieee1284.c LINK m2022_usb)
m2022_unit_test(test_usb_status test_usb_status.c LINK m2022_usb)
m2022_unit_test(test_qpdl_records test_qpdl_records.c LINK m2022_qpdl)
m2022_unit_test(test_codec11 test_codec11.c LINK m2022_qpdl)
m2022_unit_test(test_qpdl_fixtures test_qpdl_fixtures.c LINK m2022_qpdl)
target_compile_definitions(test_qpdl_fixtures PRIVATE M2022_FIXTURE_DIR="${PROJECT_SOURCE_DIR}/fixtures/oracle/samsung")
m2022_unit_test(test_media test_media.c LINK m2022_media cups)
target_compile_options(test_media PRIVATE -Wno-deprecated-declarations)
